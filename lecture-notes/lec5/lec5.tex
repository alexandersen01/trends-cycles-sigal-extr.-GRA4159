\section{Inflation targering in an SOE}
\textbf{Transmission mechanism}
\begin{itemize}
    \item The open-economy IS-curve: 
    \begin{gather*}
        y = - \alpha_1(r - \rho) + \alpha_2e + v
    \end{gather*}
    where $e = s + p^* - p $ is the real exchange rate (in logs)
    \item The open-economy Phillips curve: 
    \begin{gather*}
        \pi = \pi^e + \gamma_1y + \gamma_2 + u
    \end{gather*}
    \item The real UIP:
    \begin{gather*}
        e = e^e - (r - r^*) + z \iff r = r^* + (e^e - e) + z
    \end{gather*}
\end{itemize}

\textbf{Monetary policy}
\begin{itemize}
    \item The central bank minimizes the loss 
    \begin{gather*}
        L = \frac{1}{2}\left[(\pi - \pi^*)^2 + \lambda y^2\right]
    \end{gather*}
    \item The optimality condition is 
    \begin{gather*}
        \pi - \pi^* = -\frac{\lambda}{\gamma_1 + \frac{\gamma_2}{\alpha_1 + \alpha_2}}y
    \end{gather*}
    which again can be written as 
    \begin{gather*}
        y = -\frac{\gamma_1 + \frac{\gamma_2}{\alpha_1 + \alpha_2}}{\lambda}(\pi - \pi^*)
    \end{gather*}
    \item Reduced-form IS equation:
    \begin{gather*}
        y = -(\alpha_1 + \alpha_2)(r - \bar{r})
    \end{gather*}
    where $\bar{r} = \rho + \frac{1}{\alpha_1 + \alpha_2}v + \frac{\alpha_2}{\alpha_1 + \alpha_2}\big((r^* - \rho) + e^e + z\big) $
    \item Reduced-form Phillips curve: 
    \begin{gather*}
        \pi = \pi^2 + \left(\gamma_1 + \frac{\gamma_2}{\alpha_1 + \alpha_2}\right)y + u^{\text{open}}
    \end{gather*}
    where $u^{\text{open}} = u + \frac{\gamma_2}{\alpha_1 + \alpha_2}\big(\alpha_1(r^* - \rho + e^e + z) - v\big) $
    \item The optimality condition is 
    \begin{gather*}
        \pi - \pi^* = -\frac{\lambda}{\gamma_1 + \frac{\gamma_2}{\alpha_1 + \alpha_2}}y
    \end{gather*}
\end{itemize}
